
求解从 1 乘到 5

```c
#include <stdio.h>

int main() {
    int result = 1;  // 初始化结果为 1
    int i;
    
    // 循环从 1 到 5，计算乘积
    for (i = 1; i <= 5; i++) {
        result = result * i;  // 每次将当前数字乘到结果中
    }
    
    printf("1乘到5的结果是：%d\n", result);  // 输出结果 20
    return 0;
}
```
