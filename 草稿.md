    rewrite ^/gpustack_api/(.*)$ /$1 break;
    proxy_pass http://211.90.240.240:30001/;
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
	# SSE 支持所需的配置
	proxy_http_version 1.1;          # 使用HTTP/1.1
	proxy_set_header Connection "";  # 清除Connection头，避免被改写为close
	proxy_buffering off;             # 禁用缓冲
	proxy_cache off;                 # 禁用缓存
	proxy_read_timeout 86400s;       # 设置长超时时间（24小时）
	chunked_transfer_encoding off;   # 禁用分块传输编码
}
